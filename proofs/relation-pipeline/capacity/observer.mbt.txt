///|
/// Proof-only source overlay. No observed object is stored outside this call.
fn capacity_observe(state : BoundedBindingRowsState, event : String) -> Unit {
  if !state.distinct || state.dedupe { return }
  let n = bounded_binding_input_row_count(state.input)
  let w = state.output_names.length()
  let mut index = state.sorted_indices.map(a => a.length()).unwrap_or(0)
  // The nested sort is observed in its own state: these two arrays are still
  // reserved here even though the parent has not received their result.
  if state.sort_execution is Some(_) { index = n * 2 }
  let mut output_rows = 0
  let mut output_cells = 0
  if state.output is Some(output) {
    output_rows = output.buffer.rows.length()
    for row in output.buffer.rows { output_cells += row.length() }
  } else if state.rows.val is Some(rows) {
    output_rows = rows.rows.length()
    for row in rows.rows { output_cells += row.length() }
  }
  let metadata = state.input_names.length() + state.output_names.length() +
    state.input_ordinals.length() + state.binding_ordinals.length() +
    state.projection_ordinals.map(m => m.length()).unwrap_or(0)
  let temporary = state.snapshot.length() + state.values.length() +
    state.slot_roots.length() + state.output_row.length() +
    state.checked_row.length() + state.capture_row.length()
  let unmodeled = state.bindings.length() + state.dedupe_orders.length() +
    state.watermark_times.length() + state.lag_values.length() +
    state.json_path.map(a => a.length()).unwrap_or(0) +
    state.unique_keys.map(a => a.length()).unwrap_or(0)
  let measure_frames = state.charge_measure.map(c => c.frames.length()).unwrap_or(0) +
    state.sort_measure.map(c => c.frames.length()).unwrap_or(0) +
    state.output_charge_measure.map(c => c.frames.length()).unwrap_or(0) +
    state.capture_charge_measure.map(c => c.frames.length()).unwrap_or(0)
  println("ACTQL_PIPELINE_FAULT:{\"kind\":\"capacity observation\",\"event\":\"" + event +
    "\",\"n\":" + n.to_string() + ",\"w\":" + w.to_string() +
    ",\"live\":" + state.workspace.live.to_string() +
    ",\"limit\":" + state.workspace.limit.to_string() +
    ",\"peak\":" + state.workspace.peak.to_string() +
    ",\"metadata\":" + metadata.to_string() +
    ",\"index\":" + index.to_string() +
    ",\"keep\":" + state.distinct_keep.length().to_string() +
    ",\"terms\":" + state.sort_orders.length().to_string() +
    ",\"outputRows\":" + output_rows.to_string() +
    ",\"outputCells\":" + output_cells.to_string() +
    ",\"roots\":" + state.input_roots.length().to_string() +
    ",\"rowTemporary\":" + temporary.to_string() +
    ",\"measureFrames\":" + measure_frames.to_string() +
    ",\"unmodeledEntries\":" + unmodeled.to_string() + "}")
}
